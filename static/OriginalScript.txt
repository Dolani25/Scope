        document.addEventListener('DOMContentLoaded', () => {
            const progressBars = document.querySelectorAll('.circular-progress');
            
            progressBars.forEach(bar => {
                const rating = parseFloat(bar.dataset.rating);
                const progress = (rating / 5) * 100; // Convert rating to percentage
                bar.style.setProperty('--progress', progress);

                // Set numeric rating display
                const ratingValue = bar.querySelector('.circular-progress-value');
                ratingValue.textContent = rating.toFixed(1);
            });
        });

function updateUI(data) {

    // updating UI

    const container = $('.bottom')
    //The tab exist first in the bottom div
    

    let html = '<div class="tab"><h4 style="font-size: 83.7%; font-weight : 700;">Recent Transactions</h4><span id="airBtn" style="color : grey;" onclick="populate(this.id)">...</span></div>'



    data.forEach((item, index) => {
        let airdropPossessed = (Math.random() * 14).toFixed(8);
        
        //instantiate credibilityand sign its value.
        
        let credibility = new Intl.NumberFormat("en-US", {
    signDisplay: "exceptZero"
}).format(item.price_change_percentage_24h.toFixed
(2)); 

       
        
        let divChange = "";

        const div = `

            <div class='box' id='${index == 0 && 'wheat'}' >
            <span class="down1">
            <img src='${item.image}' class='airdrops' alt='${item.name}' /></span>
            
            
                          
              <span class="down2">
                  <h4 id="airdropName">${item.name}</h4>
                  <span id="startAmount">$${(item.current_price).toFixed(2)} <span id="credibility" onclick ="this.id = 'credibilityRed';"                    >${credibility}%</span></span>
              </span>
              <span class="down3">
                  <h4 id="airdropValue">$${(item.current_price * airdropPossessed).toFixed(2)}</h4>
                  <span id="airdropPossessed">${airdropPossessed}</span>
              </span>
          </div>       

                

        `
        
      //change price_change_percentage_24h font color to red if less than 1
      
        
      if (Math.sign(credibility)== -1){
           divChange = div.replace('id="credibility"', 'id="credibilityRed"') ;
           
           
            
        }
        else{
            divChange += div;
        }
      
      

        html += divChange;

    })
  
            

    container.innerHTML = html + '<div id="add" class="dashed-box">+ Add Wallet</div></div>'  ;
    

}
